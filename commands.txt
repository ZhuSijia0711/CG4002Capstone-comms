  // Handle new TCP client connections
  if (!tcpClientConnected) {
    tcpClient = server.available();
    if (tcpClient) {
      tcpClientConnected = true;
      Serial.println("✅ TCP Client connected");
    }
  }

  if (tcpClientConnected && tcpClient.connected()) {
    while (tcpClient.available()) {
      // Read bytes into buffer starting from current position
      int n = tcpClient.read(recvBuffer + bytesInBuffer, sizeof(recvBuffer) - bytesInBuffer);
      bytesInBuffer += n;

      // Process all complete 16-byte packets
      while (bytesInBuffer >= PACKET_SIZE) {
        byte packet[PACKET_SIZE];
        memcpy(packet, recvBuffer, PACKET_SIZE);

        // Decrypt packet using XOR key
        byte decrypted[PACKET_SIZE];
        for (int i = 0; i < PACKET_SIZE; i++) {
          decrypted[i] = packet[i] ^ xorKey[i % 16];
        }

        // Convert decrypted bytes to string command
        String message = "";
        for (int i = 0; i < PACKET_SIZE; i++) {
          if (decrypted[i] >= 32 && decrypted[i] <= 126) message += (char)decrypted[i];
          else if (decrypted[i] == 0) break;
        }
        message.trim();

        if (message.length() > 0) {
          Serial.print("📝 Decrypted message: '");
          Serial.print(message);
          Serial.println("'");
          executeCommand(message);

          // Send ACK back
          tcpClient.print("ACK:");
          tcpClient.println(message);
        }

        // Shift remaining bytes in buffer
        bytesInBuffer -= PACKET_SIZE;
        memmove(recvBuffer, recvBuffer + PACKET_SIZE, bytesInBuffer);
      }
    }
  }

  // Handle TCP client disconnection
  if (tcpClientConnected && !tcpClient.connected()) {
    tcpClient.stop();
    tcpClientConnected = false;
    Serial.println("⚠️ TCP Client disconnected");
  }
